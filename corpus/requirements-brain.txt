# Fact brain dependencies (see brain.py). Install into corpus/.venv:
#   python3 -m venv .venv && .venv/bin/pip install -r requirements-brain.txt
#
# libsql-experimental is the Python binding to libSQL, the Rust fork of SQLite with
# native vector search (F32_BLOB, vector_distance_cos, DiskANN vector_top_k). numpy
# backs the calibration/exploration math. model2vec provides static distilled
# sentence embeddings (real semantics, deterministic, CPU-only, no torch); if it is
# unavailable the brain falls back to the built-in deterministic hashing embedder, so
# the brain and its tests remain fully functional offline.

numpy>=2.0
